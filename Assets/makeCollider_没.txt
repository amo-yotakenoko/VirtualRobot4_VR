        public Vector3 v0;

        public Vector3 v2;

        public List<Triangle> neighbors = new List<Triangle>();

        public Triangle(Vector3 ve0, Vector3 ve1, Vector3 ve2)
        {
            v0 = ve0;
            v1 = ve1;
            v2 = ve2;
        }
        public IEnumerable<(Vector3, Vector3)> GetEdges()
        {
            yield return GetOrderedEdge(v0, v1);
            yield return GetOrderedEdge(v1, v2);
            yield return GetOrderedEdge(v2, v0);
        }

        private (Vector3, Vector3) GetOrderedEdge(Vector3 a, Vector3 b)
        {
            return (a.x < b.x || (a.x == b.x && a.y < b.y) || (a.x == b.x && a.y == b.y && a.z < b.z)) ? (a, b) : (b, a);
        }

        //辺が凸かどうか調べる
        public static bool IsConvex(Triangle triA, Triangle triB)
        {
            (Vector3, Vector3) sharedEdge;
            // 1. 共通エッジ（2つの共通頂点）を探す
            var sharedVertices = new HashSet<Vector3>(new[] { triA.v0, triA.v1, triA.v2 });
            sharedVertices.IntersectWith(new[] { triB.v0, triB.v1, triB.v2 });

            if (sharedVertices.Count != 2)
            {
                sharedEdge = default;
                Debug.LogWarning("隣接していない三角形です");
                return false; // 共通のエッジがない
            }

            var sharedArray = sharedVertices.ToArray();
            sharedEdge = (sharedArray[0], sharedArray[1]);

            // 2. 法線計算（ローカル空間）
            Vector3 normalA = Vector3.Cross(triA.v1 - triA.v0, triA.v2 - triA.v0).normalized;
            Vector3 normalB = Vector3.Cross(triB.v1 - triB.v0, triB.v2 - triB.v0).normalized;

            // 3. 共通エッジの方向（片方向に統一）
            Vector3 edgeDir = (sharedEdge.Item2 - sharedEdge.Item1).normalized;

            // 4. cross(normalA, normalB) と edgeDir の内積
            float dir = Vector3.Dot(Vector3.Cross(normalA, normalB), edgeDir);

            // 5. 判定（※左手系/右手系や三角形の頂点順で符号が変わる）
            return dir > 0f; // true = 凸, false = 凹
        }
    }

    Matrix4x4 transformToMatrix(Transform transform, Vector3? offset = null)
    {
        Vector3 position = transform.position;
        if (offset.HasValue)
        {
            position += offset.Value;
        }

        Vector3 scale = transform.lossyScale;
        Vector3 rotation = transform.rotation.eulerAngles;

        return Matrix4x4.TRS(position, Quaternion.Euler(rotation), scale);
    }

        var collider = obj.AddComponent<MeshCollider>();
        colliders.Add(collider);

        List<int> tmpTriangleIndices = new List<int>();
        print(triangles.Count);


        List<Triangle> addedTriangles = new List<Triangle>();
        Queue<Triangle> neighborsQueue = new Queue<Triangle>();

        neighborsQueue.Enqueue(triangles.Where(x => !x.added).OrderBy(_ => UnityEngine.Random.value).FirstOrDefault());

        while (neighborsQueue.Count > 0)
        {
            Triangle triangle = neighborsQueue.Dequeue();


            triangle.added = true;
            tmpVertices.Add(triangle.v0);
            tmpVertices.Add(triangle.v1);
            tmpVertices.Add(triangle.v2);

            int index = tmpVertices.Count - 3;
            tmpTriangleIndices.Add(index);
            tmpTriangleIndices.Add(index + 1);
            tmpTriangleIndices.Add(index + 2);

            foreach (var neighbor in triangle.neighbors.Where(x => !x.added))
            {

                if (Triangle.IsConvex(triangle, neighbor))
                {
                    neighborsQueue.Enqueue(neighbor);

                }
            }


            yield return null;
            yield return new WaitForSeconds(0.1f);



            Mesh mesh = new Mesh();
            mesh.vertices = tmpVertices.ToArray();
            mesh.triangles = tmpTriangleIndices.ToArray();
            mesh.RecalculateNormals();
            collider.sharedMesh = mesh;
            collider.convex = true;

            print("MeshCollider added to " + obj.name + " with " + triangles.Count + " triangles.");
        }




        yield return null;
    }

    public List<Triangle> GetTrianglesFromMeshCollider(GameObject obj)
    {
        List<Triangle> triangleList = new List<Triangle>();
        print("GetTrianglesFromMeshCollider called for " + obj.name);
        // MeshColliderを取得
        MeshFilter collider = obj.GetComponent<MeshFilter>();
        if (collider == null || collider.sharedMesh == null)
        {
            Debug.LogWarning("MeshCollider または Mesh が見つかりません");
            return triangleList;
        }

        Mesh mesh = collider.sharedMesh;

        Vector3[] vertices = mesh.vertices;
        int[] indices = mesh.triangles;

        print(indices.Length);

        // 3つずつインデックスを取り出して Triangle を作成
        for (int i = 0; i < indices.Length; i += 3)
        {
            print("Triangle: " + i / 3 + " - " + indices[i] + ", " + indices[i + 1] + ", " + indices[i + 2]);
            Vector3 v0 = obj.transform.TransformPoint(vertices[indices[i]]);
            Vector3 v1 = obj.transform.TransformPoint(vertices[indices[i + 1]]);
            Vector3 v2 = obj.transform.TransformPoint(vertices[indices[i + 2]]);

            Triangle tri = new Triangle(v0, v1, v2);
            triangleList.Add(tri);
        }

        BuildTriangleNeighbors(triangleList);

        return triangleList;
    }


    public void BuildTriangleNeighbors(List<Triangle> triangles)
    {
        // エッジから三角形リストを引ける辞書
        Dictionary<(Vector3, Vector3), List<Triangle>> edgeMap = new Dictionary<(Vector3, Vector3), List<Triangle>>();

        // 各三角形の各エッジを登録
        foreach (var tri in triangles)
        {
            foreach (var edge in tri.GetEdges())
            {
                if (!edgeMap.TryGetValue(edge, out var list))
                {
                    list = new List<Triangle>();
                    edgeMap[edge] = list;
                }
                list.Add(tri);
            }
        }

        // 隣接関係を登録（同じエッジを持つ三角形同士）
        foreach (var tri in triangles)
        {
            HashSet<Triangle> neighborSet = new HashSet<Triangle>();
            foreach (var edge in tri.GetEdges())
            {
                foreach (var other in edgeMap[edge])
                {
                    if (other != tri)
                    {
                        neighborSet.Add(other);
                    }
                }
            }
            tri.neighbors = neighborSet.ToList();
        }
    }
